rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Leads collection - anyone can create (booking form), only authenticated users can read/update/delete
    match /leads/{leadId} {
      allow create: if true;
      allow read, update, delete: if request.auth != null;
    }

    // Audit log - anyone can create (login tracking), only authenticated users can read
    match /audit_log/{logId} {
      allow create: if true;
      allow read: if request.auth != null;
      allow update, delete: if false;
    }
  }
}
